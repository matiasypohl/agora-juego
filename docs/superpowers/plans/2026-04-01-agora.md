# ÁGORA Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a real-time multiplayer civic education card game that runs on a classroom LAN, where a teacher moderates and students debate civic dilemmas.

**Architecture:** A single Node.js server acts as the authoritative game engine, serving static HTML/CSS/JS files and managing all game state via Socket.io. Clients are thin views that send actions and receive state updates. The server validates every action before broadcasting.

**Tech Stack:** Node.js, Express, Socket.io, Vanilla JS, HTML5, CSS3. No build tools or frameworks.

```javascript
// game/CardSystem.js
const STRATEGY_TYPES = [
  'respaldo', 'respaldo',     // 2/8 = 25%
  'censura', 'censura',       // 2/8 = 25%
  'fake_news',                // 1/8 = 12.5%
  'replica',                  // 1/8 = 12.5%
  'encuesta',                 // 1/8 = 12.5%
  'coalicion', 'escandalo', 'debate' // 3/8 = split rare
];

let cardIdCounter = 0;
function nextCardId() {
  return `card-${++cardIdCounter}`;
}

function shuffle(arr) {
  const a = [...arr];
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

/**
 * Distribute 5 agenda + 5 strategy cards among citizens.
 * Returns { [citizenId]: { agenda: [...], strategy: [...] } }
 */
function distributeCards(citizenIds, questions) {
  const hands = {};
  citizenIds.forEach(id => {
    hands[id] = { agenda: [], strategy: [] };
  });

  // 5 agenda cards: pick 5 random questions, assign to 5 random citizens
  const shuffledQuestions = shuffle(questions).slice(0, 5);
  const shuffledCitizens = shuffle(citizenIds);
  for (let i = 0; i < 5; i++) {
    const citizenId = shuffledCitizens[i % shuffledCitizens.length];
    hands[citizenId].agenda.push({
      id: nextCardId(),
      type: 'agenda',
      questionId: shuffledQuestions[i].id
    });
  }

  // 5 strategy cards: pick types from weighted pool, assign to random citizens
  const pool = shuffle(STRATEGY_TYPES).slice(0, 5);
  const shuffledForStrategy = shuffle(citizenIds);
  for (let i = 0; i < 5; i++) {
    const citizenId = shuffledForStrategy[i % shuffledForStrategy.length];
    hands[citizenId].strategy.push({
      id: nextCardId(),
      type: pool[i]
    });
  }

  return hands;
}

/**
 * Resolve all strategy card plays for a turn.
 * plays: [{ playerId, card: { type }, targetId }]
 * Returns { voteModifiers: { [candidateId]: number }, cancelled: [playerId], revealScoreboard, escandalo: targetId|null, debate: targetId|null }
 */
function resolveStrategyPhase(plays) {
  const result = {
    voteModifiers: {},
    cancelled: [],
    revealScoreboard: false,
    escandalo: null,
    debate: null,
    coalicion: null
  };

  // Separate fake_news from other plays
  const fakeNewsPlays = plays.filter(p => p.card.type === 'fake_news');
  const otherPlays = plays.filter(p => p.card.type !== 'fake_news');

  // Track which plays get cancelled by fake_news
  const cancelledPlayerIds = new Set();
  for (const fn of fakeNewsPlays) {
    // Find first uncancelled respaldo or censura targeting same candidate
    const target = otherPlays.find(
      p => (p.card.type === 'respaldo' || p.card.type === 'censura')
        && p.targetId === fn.targetId
        && !cancelledPlayerIds.has(p.playerId)
    );
    if (target) {
      cancelledPlayerIds.add(target.playerId);
    }
  }

  result.cancelled = [...cancelledPlayerIds];

  // Apply remaining effects
  for (const play of otherPlays) {
    if (cancelledPlayerIds.has(play.playerId)) continue;

    switch (play.card.type) {
      case 'respaldo':
        result.voteModifiers[play.targetId] = (result.voteModifiers[play.targetId] || 0) + 2;
        break;
      case 'censura':
        result.voteModifiers[play.targetId] = (result.voteModifiers[play.targetId] || 0) - 2;
        break;
      case 'encuesta':
        result.revealScoreboard = true;
        break;
      case 'coalicion':
        result.coalicion = { playerId: play.playerId, targetId: play.targetId };
        break;
      case 'escandalo':
        result.escandalo = play.targetId;
        break;
      case 'debate':
        result.debate = play.targetId;
        break;
      // replica is not played in strategy phase
    }
  }

  return result;
}

function resetCardIdCounter() {
  cardIdCounter = 0;
}

module.exports = { distributeCards, resolveStrategyPhase, resetCardIdCounter };

// game/RoomManager.js
const rooms = new Map();

function generateCode() {
  const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
  let code = '';
  for (let i = 0; i < 6; i++) {
    code += chars[Math.floor(Math.random() * chars.length)];
  }
  return code;
}

let playerIdCounter = 0;

function createRoom(teacherSocketId) {
  let code;
  do { code = generateCode(); } while (rooms.has(code));

  const room = {
    code,
    teacherSocketId,
    status: 'waiting', // waiting | playing | finished
    players: {},
    candidates: [],
    game: null // GameEngine state goes here
  };
  rooms.set(code, room);
  return room;
}

function getRoom(code) {
  return rooms.get(code) || null;
}

function getRoomBySocket(socketId) {
  for (const room of rooms.values()) {
    if (room.teacherSocketId === socketId) return room;
    for (const p of Object.values(room.players)) {
      if (p.socketId === socketId) return room;
    }
  }
  return null;
}

function joinRoom(code, name, socketId) {
  const room = rooms.get(code);
  if (!room) throw new Error('Room not found');
  if (room.status !== 'waiting') throw new Error('Game already in progress');

  const id = `player-${++playerIdCounter}`;
  const player = {
    id,
    name,
    socketId,
    role: 'citizen',
    hand: { agenda: [], strategy: [] },
    connected: true
  };
  room.players[id] = player;
  return player;
}

function disconnectPlayer(socketId) {
  for (const room of rooms.values()) {
    for (const p of Object.values(room.players)) {
      if (p.socketId === socketId) {
        p.connected = false;
        return { room, player: p };
      }
    }
  }
  return null;
}

function reconnectPlayer(code, playerId, newSocketId) {
  const room = rooms.get(code);
  if (!room || !room.players[playerId]) return null;
  room.players[playerId].socketId = newSocketId;
  room.players[playerId].connected = true;
  return room.players[playerId];
}

function assignRole(code, playerId, role) {
  const room = rooms.get(code);
  if (!room) throw new Error('Room not found');
  const player = room.players[playerId];
  if (!player) throw new Error('Player not found');

  // Remove from candidates if was candidate
  room.candidates = room.candidates.filter(id => id !== playerId);

  player.role = role;
  if (role === 'candidate') {
    room.candidates.push(playerId);
  }
}

function reset() {
  rooms.clear();
  playerIdCounter = 0;
}

module.exports = {
  createRoom, getRoom, getRoomBySocket, joinRoom,
  disconnectPlayer, reconnectPlayer, assignRole, reset
};

// game/GameEngine.js
const CardSystem = require('./CardSystem');

const PHASE_DURATIONS = {
  'agenda-window': 10,
  'agenda': 60,
  'replica-question': 30,
  'replica-answer': 30,
  'voting': 20,
  'strategy': 20,
  'turn-end': 5
};

function initGame(room, allQuestions) {
  const citizenIds = Object.values(room.players)
    .filter(p => p.role === 'citizen')
    .map(p => p.id);

  // Distribute cards
  const hands = CardSystem.distributeCards(citizenIds, allQuestions);
  for (const [citizenId, hand] of Object.entries(hands)) {
    room.players[citizenId].hand = hand;
  }

  // Build question bank (full list minus distributed agenda cards)
  const distributedQIds = new Set(
    Object.values(hands)
      .flatMap(h => h.agenda)
      .map(c => c.questionId)
  );
  const questionBank = allQuestions.filter(q => !distributedQIds.has(q.id));

  const scores = {};
  const questionCounts = {};
  room.candidates.forEach(id => {
    scores[id] = 0;
    questionCounts[id] = 0;
  });

  const state = {
    phase: 'agenda-window',
    currentCandidateIdx: 0,
    turn: 1,
    scores,
    questionCounts,
    questionBank,
    currentQuestion: null,
    currentTurn: freshTurn(),
    paused: false
  };

  room.game = state;
  room.status = 'playing';
  return state;
}

function freshTurn() {
  return {
    agendaClaimedBy: null,
    candidateAnswer: null,
    replicaActivatedBy: null,
    replicaQuestion: null,
    replicaAnswer: null,
    votes: {},
    strategyPlays: {},
    escandaloMultiplier: 1
  };
}

function getCurrentCandidate(room) {
  return room.candidates[room.game.currentCandidateIdx];
}

function claimAgenda(room, playerId, cardId) {
  const state = room.game;
  if (state.phase !== 'agenda-window') return { success: false, reason: 'Wrong phase' };
  if (state.currentTurn.agendaClaimedBy) return { success: false, reason: 'Already claimed' };

  const player = room.players[playerId];
  if (!player) return { success: false, reason: 'Player not found' };

  const cardIdx = player.hand.agenda.findIndex(c => c.id === cardId);
  if (cardIdx === -1) return { success: false, reason: 'Card not in hand' };

  const card = player.hand.agenda.splice(cardIdx, 1)[0];
  state.currentTurn.agendaClaimedBy = playerId;

  // Find the question
  const allQuestions = [...state.questionBank];
  // The card references a question from the original set
  // We need to find it — it might be in the bank or was already distributed
  const question = findQuestionById(room, card.questionId);
  state.currentQuestion = question;
  state.phase = 'agenda';

  return { success: true, question };
}

function findQuestionById(room, questionId) {
  // Check question bank
  const fromBank = room.game.questionBank.find(q => q.id === questionId);
  if (fromBank) return fromBank;
  // Fallback: question was distributed as agenda card, get from original data
  // This shouldn't happen normally since distributed questions are removed from bank
  return { id: questionId, text: 'Pregunta no encontrada', options: [] };
}

function autoAssignQuestion(room) {
  const state = room.game;
  if (state.questionBank.length === 0) return null;
  const idx = Math.floor(Math.random() * state.questionBank.length);
  const question = state.questionBank.splice(idx, 1)[0];
  state.currentQuestion = question;
  state.phase = 'agenda';
  return question;
}

function submitAnswer(room, playerId, answer) {
  const state = room.game;
  if (state.phase !== 'agenda') return { success: false, reason: 'Wrong phase' };
  if (getCurrentCandidate(room) !== playerId) return { success: false, reason: 'Not your turn' };
  if (state.currentTurn.candidateAnswer) return { success: false, reason: 'Already answered' };

  state.currentTurn.candidateAnswer = answer;
  return { success: true };
}

function activateReplica(room, playerId, cardId) {
  const state = room.game;
  if (state.phase !== 'agenda') return { success: false, reason: 'Wrong phase' };
  if (state.currentTurn.replicaActivatedBy) return { success: false, reason: 'Replica already used' };

  const player = room.players[playerId];
  const cardIdx = player.hand.strategy.findIndex(c => c.id === cardId && c.type === 'replica');
  if (cardIdx === -1) return { success: false, reason: 'No replica card' };

  player.hand.strategy.splice(cardIdx, 1);
  state.currentTurn.replicaActivatedBy = playerId;
  state.phase = 'replica-question';
  return { success: true };
}

function submitReplicaQuestion(room, playerId, text) {
  const state = room.game;
  if (state.phase !== 'replica-question') return { success: false, reason: 'Wrong phase' };
  if (state.currentTurn.replicaActivatedBy !== playerId) return { success: false, reason: 'Not your replica' };

  state.currentTurn.replicaQuestion = text;
  state.phase = 'replica-answer';
  return { success: true };
}

function submitReplicaAnswer(room, playerId, text) {
  const state = room.game;
  if (state.phase !== 'replica-answer') return { success: false, reason: 'Wrong phase' };
  if (getCurrentCandidate(room) !== playerId) return { success: false, reason: 'Not the candidate' };

  state.currentTurn.replicaAnswer = text;
  return { success: true };
}

function castVote(room, playerId, thumbsUp) {
  const state = room.game;
  if (state.phase !== 'voting') return { success: false, reason: 'Wrong phase' };
  if (state.currentTurn.votes[playerId] !== undefined) return { success: false, reason: 'Already voted' };

  state.currentTurn.votes[playerId] = thumbsUp;
  return { success: true };
}

function playStrategyCard(room, playerId, cardId, targetId) {
  const state = room.game;
  if (state.phase !== 'strategy') return { success: false, reason: 'Wrong phase' };
  if (state.currentTurn.strategyPlays[playerId]) return { success: false, reason: 'Already played' };

  const player = room.players[playerId];
  const cardIdx = player.hand.strategy.findIndex(c => c.id === cardId);
  if (cardIdx === -1) return { success: false, reason: 'Card not in hand' };

  const card = player.hand.strategy.splice(cardIdx, 1)[0];
  state.currentTurn.strategyPlays[playerId] = { card, targetId };
  return { success: true };
}

function sendPrivateMessage(room, playerId, candidateId, message) {
  const player = room.players[playerId];
  if (!player || player.role !== 'asesor') return { success: false, reason: 'Not an asesor' };
  // Validation only — actual message delivery is handled by socket layer
  return { success: true, candidateId, message };
}

function tallyTurn(room) {
  const state = room.game;
  const votes = state.currentTurn.votes;
  const up = Object.values(votes).filter(v => v === true).length;
  const down = Object.values(votes).filter(v => v === false).length;
  const baseVotes = up - down;

  // Resolve strategy cards
  const plays = Object.entries(state.currentTurn.strategyPlays)
    .filter(([_, play]) => play !== null)
    .map(([playerId, play]) => ({
      playerId,
      card: play.card,
      targetId: play.targetId
    }));

  const resolution = CardSystem.resolveStrategyPhase(plays);
  const candidateId = getCurrentCandidate(room);
  const modifier = resolution.voteModifiers[candidateId] || 0;
  const multiplier = state.currentTurn.escandaloMultiplier;
  const netVotes = (baseVotes + modifier) * multiplier;

  state.scores[candidateId] += netVotes;
  state.questionCounts[candidateId]++;

  // Handle coalicion: promote citizen to asesor
  if (resolution.coalicion) {
    const asesorPlayer = room.players[resolution.coalicion.playerId];
    if (asesorPlayer) asesorPlayer.role = 'asesor';
  }

  return {
    baseVotes,
    modifier,
    multiplier,
    netVotes,
    resolution,
    scores: { ...state.scores }
  };
}

function advanceTurn(room) {
  const state = room.game;
  state.currentCandidateIdx = (state.currentCandidateIdx + 1) % room.candidates.length;
  state.turn++;
  state.currentQuestion = null;
  state.currentTurn = freshTurn();
  state.phase = 'agenda-window';
}

function isGameOver(room) {
  const state = room.game;
  return room.candidates.every(id => state.questionCounts[id] >= 5);
}

function getWinner(room) {
  const state = room.game;
  let maxScore = -Infinity;
  let winnerId = null;
  for (const id of room.candidates) {
    if (state.scores[id] > maxScore) {
      maxScore = state.scores[id];
      winnerId = id;
    }
  }
  return { winnerId, scores: { ...state.scores } };
}

function setPhase(room, phase) {
  room.game.phase = phase;
}

module.exports = {
  PHASE_DURATIONS,
  initGame, freshTurn, getCurrentCandidate,
  claimAgenda, autoAssignQuestion, submitAnswer,
  activateReplica, submitReplicaQuestion, submitReplicaAnswer,
  castVote, playStrategyCard, sendPrivateMessage,
  tallyTurn, advanceTurn, isGameOver, getWinner, setPhase
};

const express = require('express');
const http = require('http');
const { Server } = require('socket.io');
const path = require('path');
const fs = require('fs');

const RoomManager = require('./game/RoomManager');
const GameEngine = require('./game/GameEngine');

const app = express();
const server = http.createServer(app);
const io = new Server(server);

// Load questions
const questionsPath = path.join(__dirname, 'data', 'questions.json');
const allQuestions = JSON.parse(fs.readFileSync(questionsPath, 'utf-8'));

// Serve static files
app.use(express.static(path.join(__dirname, 'public')));

// --- Timers ---
const roomTimers = new Map();

function clearRoomTimer(code) {
  if (roomTimers.has(code)) {
    clearTimeout(roomTimers.get(code).timeout);
    clearInterval(roomTimers.get(code).interval);
    roomTimers.delete(code);
  }
}

function startPhaseTimer(room) {
  const code = room.code;
  const phase = room.game.phase;
  const duration = GameEngine.PHASE_DURATIONS[phase];
  if (!duration) return;

  clearRoomTimer(code);

  let remaining = duration;

  const interval = setInterval(() => {
    remaining--;
    io.to(code).emit('game:timer', { phase, remaining });
    if (remaining <= 0) clearInterval(interval);
  }, 1000);

  const timeout = setTimeout(() => {
    clearInterval(interval);
    onPhaseTimeout(room);
  }, duration * 1000);

  roomTimers.set(code, { timeout, interval });
}

function onPhaseTimeout(room) {
  const state = room.game;
  if (state.paused) return;

  switch (state.phase) {
    case 'agenda-window':
      // No one claimed — auto-assign question
      const question = GameEngine.autoAssignQuestion(room);
      if (question) {
        broadcastPhase(room, { question });
        startPhaseTimer(room);
      }
      break;

    case 'agenda':
      // Time ran out — move to voting (skip replica window)
      GameEngine.setPhase(room, 'voting');
      broadcastPhase(room);
      startPhaseTimer(room);
      break;

    case 'replica-question':
      // Replica question time expired
      GameEngine.setPhase(room, 'voting');
      broadcastPhase(room);
      startPhaseTimer(room);
      break;

    case 'replica-answer':
      GameEngine.setPhase(room, 'voting');
      broadcastPhase(room);
      startPhaseTimer(room);
      break;

    case 'voting':
      // Reveal votes, move to strategy
      const votes = { ...state.currentTurn.votes };
      io.to(room.code).emit('vote:reveal', { votes });
      GameEngine.setPhase(room, 'strategy');
      broadcastPhase(room);
      startPhaseTimer(room);
      break;

    case 'strategy':
      // Resolve strategy, tally, check game over
      const tally = GameEngine.tallyTurn(room);
      io.to(room.code).emit('strategy:reveal', {
        plays: state.currentTurn.strategyPlays,
        tally
      });

      if (GameEngine.isGameOver(room)) {
        const winner = GameEngine.getWinner(room);
        room.status = 'finished';
        GameEngine.setPhase(room, 'game-over');
        io.to(room.code).emit('game:end', winner);
      } else {
        GameEngine.setPhase(room, 'turn-end');
        broadcastPhase(room, { tally });
        startPhaseTimer(room);
      }
      break;

    case 'turn-end':
      GameEngine.advanceTurn(room);
      broadcastPhase(room);
      startPhaseTimer(room);
      break;
  }
}

function broadcastPhase(room, extra = {}) {
  const state = room.game;
  io.to(room.code).emit('game:phase', {
    phase: state.phase,
    candidateId: GameEngine.getCurrentCandidate(room),
    candidateName: room.players[GameEngine.getCurrentCandidate(room)]?.name,
    question: state.currentQuestion,
    turn: state.turn,
    scores: state.scores,
    questionCounts: state.questionCounts,
    ...extra
  });
}

function broadcastPlayerList(room) {
  const players = Object.values(room.players).map(p => ({
    id: p.id, name: p.name, role: p.role, connected: p.connected
  }));
  io.to(room.code).emit('room:players', { players, candidates: room.candidates });
}

function sendHandToPlayer(room, playerId) {
  const player = room.players[playerId];
  if (!player) return;
  const socket = io.sockets.sockets.get(player.socketId);
  if (socket) {
    socket.emit('player:hand', { hand: player.hand });
  }
}

// --- Socket.io ---
io.on('connection', (socket) => {

  // TEACHER: Create room
  socket.on('teacher:create', (callback) => {
    const room = RoomManager.createRoom(socket.id);
    socket.join(room.code);
    socket.data = { roomCode: room.code, role: 'teacher' };
    callback({ code: room.code });
  });

  // PLAYER: Join room
  socket.on('player:join', ({ code, name }, callback) => {
    try {
      const room = RoomManager.getRoom(code);
      if (!room) return callback({ error: 'Sala no encontrada' });

      const player = RoomManager.joinRoom(code, name, socket.id);
      socket.join(code);
      socket.data = { roomCode: code, playerId: player.id, role: 'player' };
      broadcastPlayerList(room);
      callback({ playerId: player.id, name: player.name });
    } catch (err) {
      callback({ error: err.message });
    }
  });

  // TEACHER: Assign role
  socket.on('teacher:assign-role', ({ playerId, role }) => {
    const room = RoomManager.getRoom(socket.data?.roomCode);
    if (!room) return;
    RoomManager.assignRole(room.code, playerId, role);
    broadcastPlayerList(room);
  });

  // TEACHER: Start game
  socket.on('teacher:start', () => {
    const room = RoomManager.getRoom(socket.data?.roomCode);
    if (!room || room.candidates.length < 2) return;

    GameEngine.initGame(room, allQuestions);

    // Send each citizen their hand
    Object.values(room.players).forEach(p => {
      if (p.role === 'citizen') sendHandToPlayer(room, p.id);
    });

    broadcastPhase(room);
    startPhaseTimer(room);
  });

  // CITIZEN: Claim agenda card
  socket.on('agenda:claim', ({ cardId }) => {
    const room = RoomManager.getRoom(socket.data?.roomCode);
    if (!room?.game) return;

    const result = GameEngine.claimAgenda(room, socket.data.playerId, cardId);
    if (result.success) {
      clearRoomTimer(room.code);
      broadcastPhase(room, { claimedBy: socket.data.playerId });
      startPhaseTimer(room);
    }
  });

  // CANDIDATE: Submit answer
  socket.on('event:answer', ({ answer }) => {
    const room = RoomManager.getRoom(socket.data?.roomCode);
    if (!room?.game) return;

    const result = GameEngine.submitAnswer(room, socket.data.playerId, answer);
    if (result.success) {
      io.to(room.code).emit('event:answered', {
        candidateId: socket.data.playerId,
        answer
      });
    }
  });

  // CITIZEN: Activate replica
  socket.on('replica:activate', ({ cardId }) => {
    const room = RoomManager.getRoom(socket.data?.roomCode);
    if (!room?.game) return;

    const result = GameEngine.activateReplica(room, socket.data.playerId, cardId);
    if (result.success) {
      clearRoomTimer(room.code);
      broadcastPhase(room);
      startPhaseTimer(room);
      sendHandToPlayer(room, socket.data.playerId);
    }
  });

  // CITIZEN: Submit replica question
  socket.on('replica:question', ({ text }) => {
    const room = RoomManager.getRoom(socket.data?.roomCode);
    if (!room?.game) return;

    const result = GameEngine.submitReplicaQuestion(room, socket.data.playerId, text);
    if (result.success) {
      clearRoomTimer(room.code);
      io.to(room.code).emit('replica:show-question', { text, from: room.players[socket.data.playerId]?.name });
      broadcastPhase(room);
      startPhaseTimer(room);
    }
  });

  // CANDIDATE: Submit replica answer
  socket.on('replica:answer', ({ text }) => {
    const room = RoomManager.getRoom(socket.data?.roomCode);
    if (!room?.game) return;

    const result = GameEngine.submitReplicaAnswer(room, socket.data.playerId, text);
    if (result.success) {
      io.to(room.code).emit('replica:show-answer', { text });
      // Move to voting after replica answer
      clearRoomTimer(room.code);
      GameEngine.setPhase(room, 'voting');
      broadcastPhase(room);
      startPhaseTimer(room);
    }
  });

  // ALL: Cast vote
  socket.on('vote:cast', ({ thumbsUp }) => {
    const room = RoomManager.getRoom(socket.data?.roomCode);
    if (!room?.game) return;

    GameEngine.castVote(room, socket.data.playerId, thumbsUp);

    // Check if all connected players have voted
    const connected = Object.values(room.players).filter(p => p.connected);
    const voted = Object.keys(room.game.currentTurn.votes).length;
    if (voted >= connected.length) {
      // All voted — skip timer and advance
      clearRoomTimer(room.code);
      onPhaseTimeout(room);
    }
  });

  // CITIZEN: Play strategy card
  socket.on('strategy:play', ({ cardId, targetId }) => {
    const room = RoomManager.getRoom(socket.data?.roomCode);
    if (!room?.game) return;

    const result = GameEngine.playStrategyCard(room, socket.data.playerId, cardId, targetId);
    if (result.success) {
      sendHandToPlayer(room, socket.data.playerId);
    }
  });

  // ASESOR: Private message to candidate
  socket.on('message:private', ({ candidateId, message }) => {
    const room = RoomManager.getRoom(socket.data?.roomCode);
    if (!room?.game) return;

    const result = GameEngine.sendPrivateMessage(room, socket.data.playerId, candidateId, message);
    if (result.success) {
      const candidate = room.players[candidateId];
      if (candidate) {
        const candSocket = io.sockets.sockets.get(candidate.socketId);
        if (candSocket) {
          candSocket.emit('message:private', {
            from: room.players[socket.data.playerId]?.name,
            message
          });
        }
      }
    }
  });

  // TEACHER: Controls
  socket.on('teacher:control', ({ action }) => {
    const room = RoomManager.getRoom(socket.data?.roomCode);
    if (!room?.game) return;

    switch (action) {
      case 'pause':
        room.game.paused = true;
        clearRoomTimer(room.code);
        io.to(room.code).emit('game:paused', { paused: true });
        break;
      case 'resume':
        room.game.paused = false;
        io.to(room.code).emit('game:paused', { paused: false });
        startPhaseTimer(room);
        break;
      case 'skip':
        clearRoomTimer(room.code);
        onPhaseTimeout(room);
        break;
      case 'end':
        clearRoomTimer(room.code);
        const winner = GameEngine.getWinner(room);
        room.status = 'finished';
        GameEngine.setPhase(room, 'game-over');
        io.to(room.code).emit('game:end', winner);
        break;
    }
  });

  // Disconnect
  socket.on('disconnect', () => {
    const info = RoomManager.disconnectPlayer(socket.id);
    if (info) {
      broadcastPlayerList(info.room);
    }
  });
});

// Start
const PORT = process.env.PORT || 3000;
server.listen(PORT, '0.0.0.0', () => {
  const nets = require('os').networkInterfaces();
  let localIp = 'localhost';
  for (const iface of Object.values(nets)) {
    for (const addr of iface) {
      if (addr.family === 'IPv4' && !addr.internal) {
        localIp = addr.address;
        break;
      }
    }
  }

  console.log(`  ================================`);
  console.log(`  Local:    http://localhost:${PORT}`);

  console.log(`  Profesor: http://${localIp}:${PORT}/teacher.html\n`);

// public/js/game.js
(function() {
  const playerId = sessionStorage.getItem('playerId');
  const playerName = sessionStorage.getItem('playerName');
  if (!playerId) { window.location.href = '/'; return; }

  let myRole = 'citizen';
  let myHand = { agenda: [], strategy: [] };
  let currentCandidateId = null;
  let selectedStrategyCard = null;

  const CARD_META = {
    respaldo:  { icon: '🛡️', name: 'Respaldo',  desc: '+2 votos', color: 'var(--green)' },
    censura:   { icon: '⚡', name: 'Censura',   desc: '-2 votos', color: 'var(--red)' },
    fake_news: { icon: '📰', name: 'Fake News', desc: 'Anular carta', color: 'var(--orange)' },
    replica:   { icon: '💬', name: 'Réplica',   desc: 'Contra-pregunta', color: 'var(--teal)' },
    encuesta:  { icon: '📊', name: 'Encuesta',  desc: 'Ver marcador', color: 'var(--orange)' },
    coalicion: { icon: '🤝', name: 'Coalición', desc: 'Ser asesor', color: 'var(--blue)' },
    escandalo: { icon: '🔥', name: 'Escándalo', desc: 'Votos ×2', color: 'var(--red)' },
    debate:    { icon: '🤲', name: 'Debate',    desc: '2 vs 2', color: 'var(--primary-light)' },
  };

  // --- Helpers ---
  function $(id) { return document.getElementById(id); }

  function showPhase(phaseId) {
    document.querySelectorAll('[id^="phase-"]').forEach(el => el.classList.add('hidden'));
    const el = $(phaseId);
    if (el) { el.classList.remove('hidden'); el.classList.add('fade-in'); }
  }

  function formatTime(s) {
    const m = Math.floor(s / 60);
    const sec = s % 60;
    return `${m}:${sec.toString().padStart(2, '0')}`;
  }

  function updateScoreboard(scores, players) {
    const sb = $('scoreboard');
    if (!scores) return;
    sb.innerHTML = Object.entries(scores).map(([id, score]) => {
      const name = players?.[id] || id;
      return `<div class="score-row">
        <span class="name">${name}</span>
        <span class="score" style="color:var(--accent);">${score} pts</span>
      </div>`;
    }).join('');
  }

  function renderAgendaCards(cards) {
    const container = $('aw-agenda-cards');
    container.innerHTML = '';
    cards.forEach(card => {
      const btn = document.createElement('button');
      btn.className = 'btn btn-primary btn-sm';
      btn.textContent = '📋 Jugar Agenda';
      btn.onclick = () => {
        socket.emit('agenda:claim', { cardId: card.id });
        btn.disabled = true;
      };
      container.appendChild(btn);
    });
    if (cards.length === 0) {
      container.innerHTML = '<p class="subtitle">No tienes Cartas de Agenda</p>';
    }
  }

  function renderOptions(options, isCandidate) {
    const container = $('agenda-options');
    container.innerHTML = '';
    options.forEach(opt => {
      const btn = document.createElement('button');
      btn.className = 'option-btn';
      btn.innerHTML = `<span class="option-label">${opt.label}</span><span>${opt.text}</span>`;
      if (isCandidate) {
        btn.onclick = () => {
          container.querySelectorAll('.option-btn').forEach(b => b.classList.remove('selected'));
          btn.classList.add('selected');
          socket.emit('event:answer', { answer: opt.label });
        };
      } else {
        btn.style.cursor = 'default';
      }
      container.appendChild(btn);
    });
  }

  function renderStrategyHand() {
    const container = $('strategy-hand');
    container.innerHTML = '';
    selectedStrategyCard = null;
    $('btn-play-card').classList.add('hidden');

    myHand.strategy.forEach(card => {
      const meta = CARD_META[card.type] || { icon: '🃏', name: card.type, desc: '', color: 'white' };
      const el = document.createElement('div');
      el.className = `game-card card-${card.type}`;
      el.innerHTML = `<div class="icon">${meta.icon}</div><div class="name">${meta.name}</div><div class="desc">${meta.desc}</div>`;
      el.onclick = () => {
        container.querySelectorAll('.game-card').forEach(c => c.classList.remove('selected'));
        el.classList.add('selected');
        selectedStrategyCard = card;
        $('btn-play-card').classList.remove('hidden');
      };
      container.appendChild(el);
    });
  }

  // --- Socket listeners ---

  socket.on('player:hand', (data) => {
    myHand = data.hand;
  });

  socket.on('room:players', (data) => {
    const me = data.players.find(p => p.id === playerId);
    if (me) myRole = me.role;
  });

  let playerNames = {};

  socket.on('game:phase', (data) => {
    const { phase, candidateId, candidateName, question, turn, scores, questionCounts } = data;
    currentCandidateId = candidateId;

    $('header-turn').textContent = `Turno ${turn}`;
    $('header-phase').textContent = phase.replace(/-/g, ' ').toUpperCase();

    // Build player name map from scores
    if (candidateId && candidateName) playerNames[candidateId] = candidateName;
    updateScoreboard(scores, playerNames);

    switch (phase) {
      case 'agenda-window':
        showPhase('phase-agenda-window');
        $('aw-candidate').textContent = candidateName || 'Candidato';
        renderAgendaCards(myHand.agenda);
        break;

      case 'agenda':
        showPhase('phase-agenda');
        if (question) {
          $('agenda-question').textContent = question.text;
          const isCandidate = (playerId === candidateId);
          renderOptions(question.options, isCandidate);
          $('agenda-waiting').classList.toggle('hidden', isCandidate);
          $('agenda-answered').classList.add('hidden');

          // Show replica button if citizen has replica card
          const hasReplica = myHand.strategy.some(c => c.type === 'replica');
          if (hasReplica && myRole !== 'candidate') {
            $('replica-trigger').classList.remove('hidden');
          } else {
            $('replica-trigger').classList.add('hidden');
          }
        }
        // Show asesor panel if applicable
        if (myRole === 'asesor') {
          $('asesor-panel').classList.remove('hidden');
        }
        break;

      case 'replica-question':
        showPhase('phase-replica-question');
        // The citizen who activated replica writes the question
        // Others wait
        $('replica-q-write').classList.add('hidden');
        $('replica-q-wait').classList.remove('hidden');
        break;

      case 'replica-answer':
        showPhase('phase-replica-answer');
        $('replica-a-write').classList.toggle('hidden', playerId !== candidateId);
        $('replica-a-wait').classList.toggle('hidden', playerId === candidateId);
        break;

      case 'voting':
        showPhase('phase-voting');
        $('vote-buttons').classList.remove('hidden');
        $('vote-waiting').classList.add('hidden');
        document.querySelectorAll('.vote-btn').forEach(b => b.classList.remove('voted'));
        break;

      case 'strategy':
        showPhase('phase-strategy');
        renderStrategyHand();
        $('strategy-waiting').classList.add('hidden');
        $('btn-pass').classList.remove('hidden');
        break;

      case 'turn-end':
        showPhase('phase-turn-end');
        if (data.tally) {
          $('turn-result').innerHTML = `
            <p>Votos: <strong>${data.tally.baseVotes > 0 ? '+' : ''}${data.tally.baseVotes}</strong></p>
            ${data.tally.modifier !== 0 ? `<p>Cartas: <strong>${data.tally.modifier > 0 ? '+' : ''}${data.tally.modifier}</strong></p>` : ''}
            ${data.tally.multiplier > 1 ? `<p>Escándalo: <strong>×${data.tally.multiplier}</strong></p>` : ''}
            <p style="font-size:1.3rem; color:var(--accent); margin-top:8px;">Neto: <strong>${data.tally.netVotes > 0 ? '+' : ''}${data.tally.netVotes}</strong></p>
          `;
        }
        break;
    }
  });

  socket.on('game:timer', ({ remaining }) => {
    $('header-timer').textContent = formatTime(remaining);
    $('header-timer').classList.toggle('urgent', remaining <= 5);
  });

  socket.on('event:answered', ({ candidateId: cid, answer }) => {
    if (playerId !== cid) {
      $('agenda-waiting').classList.add('hidden');
      $('agenda-answered').classList.remove('hidden');
      const options = document.querySelectorAll('.option-btn');
      options.forEach(btn => {
        if (btn.querySelector('.option-label').textContent === answer) {
          btn.classList.add('selected');
        }
      });
      $('agenda-answer-display').textContent = `Alternativa ${answer}`;
    }
  });

  socket.on('replica:show-question', ({ text, from }) => {
    $('replica-question-text').textContent = `"${text}" — ${from}`;
  });

  socket.on('replica:show-answer', ({ text }) => {
    $('replica-a-wait').innerHTML = `<p style="margin-top:8px;">${text}</p>`;
    $('replica-a-wait').classList.remove('hidden');
  });

  socket.on('vote:reveal', ({ votes }) => {
    const up = Object.values(votes).filter(v => v).length;
    const down = Object.values(votes).filter(v => !v).length;
    $('vote-buttons').innerHTML = `
      <div class="flip-in" style="text-align:center;">
        <div style="font-size:2rem;">👍 ${up}</div>
      </div>
      <div class="flip-in" style="text-align:center;">
        <div style="font-size:2rem;">👎 ${down}</div>
      </div>
    `;
  });

  socket.on('strategy:reveal', ({ plays, tally }) => {
    // Update scoreboard with new scores
    if (tally?.scores) updateScoreboard(tally.scores, playerNames);
  });

  socket.on('game:end', ({ winnerId, scores }) => {
    showPhase('phase-game-over');
    $('winner-name').textContent = `¡${playerNames[winnerId] || 'Ganador'} es ELEGIDO/A!`;
    $('final-scores').innerHTML = Object.entries(scores)
      .sort((a, b) => b[1] - a[1])
      .map(([id, s]) => `<div class="score-row"><span class="name">${playerNames[id] || id}</span><span class="score">${s} pts</span></div>`)
      .join('');
  });

  socket.on('message:private', ({ from, message }) => {
    $('private-msg-box').classList.remove('hidden');
    $('private-msg-text').textContent = `${from}: "${message}"`;
    setTimeout(() => $('private-msg-box').classList.add('hidden'), 8000);
  });

  socket.on('game:paused', ({ paused }) => {
    $('header-phase').textContent = paused ? '⏸ PAUSADO' : $('header-phase').textContent;
  });

  // --- UI Event Listeners ---

  // Vote buttons
  document.querySelectorAll('.vote-btn').forEach(btn => {
    btn.addEventListener('click', () => {
      const thumbsUp = btn.dataset.vote === 'up';
      socket.emit('vote:cast', { thumbsUp });
      document.querySelectorAll('.vote-btn').forEach(b => b.classList.add('hidden'));
      btn.classList.remove('hidden');
      btn.classList.add('voted');
      $('vote-waiting').classList.remove('hidden');
    });
  });

  // Replica trigger
  $('btn-replica')?.addEventListener('click', () => {
    const replicaCard = myHand.strategy.find(c => c.type === 'replica');
    if (replicaCard) {
      socket.emit('replica:activate', { cardId: replicaCard.id });
      $('btn-replica').disabled = true;
    }
  });

  // Replica question submit
  $('btn-replica-q')?.addEventListener('click', () => {
    const text = $('replica-q-input').value.trim();
    if (text) {
      socket.emit('replica:question', { text });
      $('btn-replica-q').disabled = true;
    }
  });

  // Replica answer submit
  $('btn-replica-a')?.addEventListener('click', () => {
    const text = $('replica-a-input').value.trim();
    if (text) {
      socket.emit('replica:answer', { text });
      $('btn-replica-a').disabled = true;
    }
  });

  // Strategy: Pass
  $('btn-pass')?.addEventListener('click', () => {
    $('strategy-waiting').classList.remove('hidden');
    $('btn-pass').classList.add('hidden');
    $('btn-play-card').classList.add('hidden');
    $('strategy-hand').innerHTML = '<p class="subtitle">Pasaste este turno</p>';
  });

  // Strategy: Play selected card
  $('btn-play-card')?.addEventListener('click', () => {
    if (!selectedStrategyCard) return;
    // For targeted cards, target the current candidate
    socket.emit('strategy:play', {
      cardId: selectedStrategyCard.id,
      targetId: currentCandidateId
    });
    $('strategy-waiting').classList.remove('hidden');
    $('btn-pass').classList.add('hidden');
    $('btn-play-card').classList.add('hidden');
    $('strategy-hand').innerHTML = '<p class="subtitle">Carta jugada. Esperando revelación...</p>';
  });

  // Asesor: Send private message
  $('btn-asesor-send')?.addEventListener('click', () => {
    const msg = $('asesor-msg').value.trim();
    if (msg && currentCandidateId) {
      socket.emit('message:private', { candidateId: currentCandidateId, message: msg });
      $('asesor-msg').value = '';
    }
  });

  // Listen for replica-question phase to show write box to the right player
  socket.on('game:phase', (data) => {
    if (data.phase === 'replica-question') {
      // Check if we activated the replica
      const isReplicaOwner = myHand.strategy.some(c => c.type === 'replica') === false; // card was consumed
      // Simpler: server should tell us. For now, show write to all citizens and let server reject
      if (myRole !== 'candidate') {
        $('replica-q-write').classList.remove('hidden');
        $('replica-q-wait').classList.add('hidden');
      }
    }
  });

})();

        <span style="flex:1; font-size:0.88rem;">${p.name}</span>

            ${p.role === 'candidate' ? '→ Ciudadano' : '→ Candidato'}
          </button>
        ` : ''}
      </div>
    `).join('');
  });

  window.toggleRole = (playerId, currentRole) => {

  };

  // Start game
  $('btn-start').addEventListener('click', () => {

    $('btn-end').classList.remove('hidden');
  });

  // Controls
  $('btn-pause').addEventListener('click', () => {
    socket.emit('teacher:control', { action: 'pause' });
    $('btn-pause').classList.add('hidden');
    $('btn-resume').classList.remove('hidden');
  });
  $('btn-resume').addEventListener('click', () => {
    socket.emit('teacher:control', { action: 'resume' });
    $('btn-resume').classList.add('hidden');
    $('btn-pause').classList.remove('hidden');

    socket.emit('teacher:control', { action: 'skip' });

      socket.emit('teacher:control', { action: 'end' });
    }
  });

    // Scoreboard

      $('t-scoreboard').innerHTML = Object.entries(data.scores).map(([id, score]) =>
        `<div class="score-row">
          <span class="name">${playerNames[id] || id}</span>
          <span class="score" style="color:var(--accent);">${score} pts</span>
        </div>`
      ).join('');
    }

    // Progress
    if (data.questionCounts) {
      $('t-progress').innerHTML = '<div class="label" style="margin-bottom:6px;">Progreso</div>' +

            <div class="flex-between" style="margin-bottom:3px;">
              <span style="font-size:0.8rem;">${playerNames[id] || id}</span>
              <span style="font-size:0.75rem; color:var(--text-dim);">${count}/5</span>
            </div>
            <div class="progress-bar"><div class="fill" style="width:${pct}%; background:var(--primary);"></div></div>
          </div>`;
        }).join('');
    }

    // Current question
    if (data.question) {
      $('question-section').classList.remove('hidden');
      $('t-question').textContent = data.question.text;
      $('t-answer').textContent = '';
    }
  });

  socket.on('event:answered', ({ candidateId, answer }) => {
    $('t-answer').textContent = `${playerNames[candidateId] || 'Candidato'} eligió: ${answer}`;
  });

  socket.on('game:timer', ({ remaining }) => {
    const m = Math.floor(remaining / 60);
    const s = remaining % 60;
    $('t-timer').textContent = `${m}:${s.toString().padStart(2, '0')}`;
    $('t-timer').classList.toggle('urgent', remaining <= 5);
  });

  socket.on('game:end', ({ winnerId, scores }) => {
    $('t-phase').textContent = '🏆 FIN';
    $('btn-pause').classList.add('hidden');
    $('btn-resume').classList.add('hidden');
    $('btn-skip').classList.add('hidden');
    $('btn-end').classList.add('hidden');
    $('t-scoreboard').innerHTML = `
      <div style="text-align:center; padding:16px;">
        <h2>🏆 ${playerNames[winnerId] || 'Ganador'} es ELEGIDO/A</h2>
      </div>` +
      Object.entries(scores)
        .sort((a, b) => b[1] - a[1])
        .map(([id, s]) =>
          `<div class="score-row"><span class="name">${playerNames[id] || id}</span><span class="score">${s} pts</span></div>`
        ).join('');
  });

  socket.on('game:paused', ({ paused }) => {
    $('t-phase').textContent = paused ? '⏸ PAUSADO' : '';
  });

})();
```
